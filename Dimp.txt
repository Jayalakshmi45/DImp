Exp 7
sudo apt update
sudo apt upgrade -y
sudo apt install ansible -y
sudo ansible-playbook -i hosts.ini setup.yml 


1.hosts.ini
[local]
localhost ansible_connection=local

2.setup.yml
---
- name: Basic Server Setup
  hosts: localhost
  become: yes
  tasks:
  - name: Update apt cache
    apt:
      update_cache: yes
  - name: Install Apache
    apt:
      name: apache2
      state: present
  - name: Start Apache service
    service:
      name: apache2
      state: started
      enabled: yes
  - name: Create web page
    copy:
      dest: /var/www/html/index.html
      content: "<h1>Ansible Web Server Working</h1>"
